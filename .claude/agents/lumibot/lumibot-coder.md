def _get_target_option_contract(self):
    """
    Find and return the target option contract based on strategy rules.

    Returns:
        Asset object for the target option, or None if not found
    """
    try:
        # Get option chains for the underlying
        chains = self.get_chains(Asset(self.symbol))

        if chains is None:
            self.log_message("No option chains available")
            return None

        # Get expirations for the desired option type
        option_type = self.parameters.get("option_type", "PUT")  # PUT or CALL
        expiries = chains.expirations(option_type)

        if not expiries:
            self.log_message(f"No {option_type} expirations available")
            return None

        # Filter for target DTE (Days To Expiration)
        target_dte = self.parameters.get("dte_target", 0)  # 0 for 0DTE
        today = date.today()

        target_expiries = [
            exp for exp in expiries
            if (exp - today).days == target_dte
        ]

        if not target_expiries:
            self.log_message(f"No expirations with {target_dte} DTE found")
            return None

        expiry = target_expiries[0]

        # Get strikes for this expiration
        strikes = chains.strikes(expiry, option_type)

        if not strikes:
            self.log_message(f"No strikes available for {expiry}")
            return None

        # Find target strike based on strategy
        current_price = self.get_last_price(self.symbol)
        target_strike = self._select_strike(strikes, current_price, option_type)

        if target_strike is None:
            self.log_message("No suitable strike found")
            return None

        # Create option Asset
        option = Asset(
            symbol=self.symbol,
            asset_type=Asset.AssetType.OPTION,
            expiration=expiry,
            strike=target_strike,
            right=Asset.OptionRight.PUT if option_type == "PUT" else Asset.OptionRight.CALL
        )

        self.log_message(
            f"Selected option: {self.symbol} {expiry} {target_strike} {option_type}"
        )

        return option

    except Exception as e:
        self.log_message(f"Error getting option contract: {str(e)}")
        return None

def _select_strike(self, strikes, current_price, option_type):
    """
    Select appropriate strike price based on strategy rules.

    Args:
        strikes: List of available strikes
        current_price: Current price of underlying
        option_type: "PUT" or "CALL"

    Returns:
        Selected strike price or None
    """
    # [Implement strike selection logic from architecture]
    # Example: OTM by $5

    if option_type == "PUT":
        # For puts, OTM is below current price
        otm_strikes = [s for s in strikes if s < current_price - 5]
        if otm_strikes:
            return max(otm_strikes)  # Closest to current price
    else:
        # For calls, OTM is above current price
        otm_strikes = [s for s in strikes if s > current_price + 5]
        if otm_strikes:
            return min(otm_strikes)  # Closest to current price

    return None

def _execute_option_entry(self, option_asset, is_sell_to_open=False):
    """
    Execute option trade (buy or sell to open).

    Args:
        option_asset: Asset object for the option
        is_sell_to_open: If True, sell option (collect premium).
                        If False, buy option (pay premium).
    """
    # Calculate number of contracts
    num_contracts = self.parameters.get("num_contracts", 1)

    # Create order
    order = self.create_order(
        option_asset,
        quantity=num_contracts,
        side="sell" if is_sell_to_open else "buy"
    )

    # Submit order
    self.submit_order(order)

    # Log
    action = "SOLD" if is_sell_to_open else "BOUGHT"
    self.log_message(
        f"✅ {action} {num_contracts} contract(s): "
        f"{option_asset.symbol} {option_asset.strike} {option_asset.right}"
    )

    # Update state
    self.entry_time = self.get_datetime()
    self.position_type = "short" if is_sell_to_open else "long"
